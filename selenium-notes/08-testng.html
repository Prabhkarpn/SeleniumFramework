🎯 What is TestNG?

TestNG is a testing framework inspired by JUnit but with more powerful features. It's designed to cover all categories of tests: unit, functional, end-to-end, integration, etc.

📋 TestNG Annotations

AnnotationDescriptionExecution Order
@BeforeSuiteRuns once before all tests in the suite1
@BeforeTestRuns before any test method in <test> tag2
@BeforeClassRuns once before the first test in class3
@BeforeMethodRuns before each @Test method4
@TestMarks a method as test method5
@AfterMethodRuns after each @Test method6
@AfterClassRuns once after all tests in class7
@AfterTestRuns after all test methods in <test>8
@AfterSuiteRuns once after all tests in suite9

💻 TestNG Example

import org.testng.annotations.*;

public class LoginTest {
    
    @BeforeSuite
    public void beforeSuite() {
        System.out.println("Before Suite");
    }
    
    @BeforeTest
    public void beforeTest() {
        System.out.println("Before Test");
    }
    
    @BeforeClass
    public void beforeClass() {
        System.out.println("Before Class");
    }
    
    @BeforeMethod
    public void beforeMethod() {
        System.out.println("Before Method");
    }
    
    @Test(priority = 1)
    public void testLogin() {
        System.out.println("Test Login");
    }
    
    @Test(priority = 2)
    public void testLogout() {
        System.out.println("Test Logout");
    }
    
    @AfterMethod
    public void afterMethod() {
        System.out.println("After Method");
    }
    
    @AfterClass
    public void afterClass() {
        System.out.println("After Class");
    }
    
    @AfterTest
    public void afterTest() {
        System.out.println("After Test");
    }
    
    @AfterSuite
    public void afterSuite() {
        System.out.println("After Suite");
    }
}

🔍 @Test Attributes

// Priority (execution order)
@Test(priority = 1)
public void firstTest() { }

// Description
@Test(description = "Verify login functionality")
public void testLogin() { }

// Enabled/Disabled
@Test(enabled = false)
public void skipThisTest() { }

// Expected exception
@Test(expectedExceptions = ArithmeticException.class)
public void testException() {
    int i = 1/0;
}

// Timeout
@Test(timeOut = 5000)  // milliseconds
public void testWithTimeout() { }

// Depends on other tests
@Test(dependsOnMethods = {"testLogin"})
public void testProfile() { }

// Groups
@Test(groups = {"smoke", "regression"})
public void testSearch() { }

✅ TestNG Assertions

import org.testng.Assert;

// Equality assertions
Assert.assertEquals(actual, expected);
Assert.assertEquals(actual, expected, "Error message");

// True/False assertions
Assert.assertTrue(condition);
Assert.assertFalse(condition);

// Null assertions
Assert.assertNull(object);
Assert.assertNotNull(object);

// Fail test
Assert.fail("Test failed");

// Soft assertions (continue on failure)
SoftAssert softAssert = new SoftAssert();
softAssert.assertEquals(actual1, expected1);
softAssert.assertEquals(actual2, expected2);
softAssert.assertAll();  // Report all failures

📄 testng.xml Configuration

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE suite SYSTEM "https://testng.org/testng-1.0.dtd">
<suite name="Test Suite" parallel="tests" thread-count="2">
    
    <test name="Login Tests">
        <classes>
            <class name="com.orangehrm.test.LoginPageTest"/>
        </classes>
    </test>
    
    <test name="Home Tests">
        <classes>
            <class name="com.orangehrm.test.HomePageTest"/>
        </classes>
    </test>
    
    <!-- Run specific groups -->
    <test name="Smoke Tests">
        <groups>
            <run>
                <include name="smoke"/>
            </run>
        </groups>
        <packages>
            <package name="com.orangehrm.test.*"/>
        </packages>
    </test>
    
</suite>

⚡ Parallel Execution

<!-- Parallel at test level -->
<suite name="Suite" parallel="tests" thread-count="3">

<!-- Parallel at class level -->
<suite name="Suite" parallel="classes" thread-count="3">

<!-- Parallel at method level -->
<suite name="Suite" parallel="methods" thread-count="5">

📊 Data Providers

@DataProvider(name = "loginData")
public Object[][] getData() {
    return new Object[][] {
        {"user1", "pass1"},
        {"user2", "pass2"},
        {"user3", "pass3"}
    };
}

@Test(dataProvider = "loginData")
public void testLogin(String username, String password) {
    System.out.println("Testing: " + username + "/" + password);
}

👂 TestNG Listeners

import org.testng.ITestListener;
import org.testng.ITestResult;

public class TestListener implements ITestListener {
    
    public void onTestStart(ITestResult result) {
        System.out.println("Test started: " + result.getName());
    }
    
    public void onTestSuccess(ITestResult result) {
        System.out.println("Test passed: " + result.getName());
    }
    
    public void onTestFailure(ITestResult result) {
        System.out.println("Test failed: " + result.getName());
        // Take screenshot on failure
    }
    
    public void onTestSkipped(ITestResult result) {
        System.out.println("Test skipped: " + result.getName());
    }
}

// Add to testng.xml
<listeners>
    <listener class-name="com.orangehrm.listeners.TestListener"/>
</listeners>

TestNG provides powerful features for organizing and executing tests, including annotations, assertions, data providers, parallel execution, and listeners.
